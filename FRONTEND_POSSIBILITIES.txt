```typescript
import axios, { AxiosInstance, AxiosError } from 'axios';
import {
  Transaction,
  PredictionResponse,
  BatchPredictionResponse,
  HealthResponse,
  ModelInfo,
  ApiError,
} from '../types/fraud';

const API_BASE_URL = process.env.REACT_APP_API_URL || 'http://127.0.0.1:8000';

// Create Axios Instance with default config
const apiClient: AxiosInstance = axios.create({
  baseURL: API_BASE_URL,
  headers: {
    'Content-Type': 'application/json',
  },
  timeout: 10000, // 10 second timeout
});

// Request Interceptor
apiClient.interceptors.request.use(
  (config) => {
    // Add auth token if available
    const token = localStorage.getItem('authToken');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    console.log(`[API] ${config.method?.toUpperCase()} ${config.url}`);
    return config;
  },
  (error) => {
    console.error('[API] Request error:', error);
    return Promise.reject(error);
  }
);

// Response Interceptor for error handling
apiClient.interceptors.response.use(
  (response) => {
    console.log(`[API] Response OK: ${response.status}`);
    return response;
  },
  (error: AxiosError) => {
    const apiError: ApiError = {
      status: error.response?.status || 500,
      message: error.message,
      details: (error.response?.data as any)?.detail || error.response?.statusText,
    };
    console.error('[API] Response error:', apiError);
    return Promise.reject(apiError);
  }
);

// Fraud Detection API Functions
export const fraudApi = {
  // Check API health
  async health(): Promise<HealthResponse> {
    const response = await apiClient.get<HealthResponse>('/health');
    return response.data;
  },

  // Single transaction prediction
  async predict(transaction: Transaction): Promise<PredictionResponse> {
    const response = await apiClient.post<PredictionResponse>('/predict', transaction);
    return response.data;
  },

  // Batch predictions for multiple transactions
  async predictBatch(transactions: Transaction[]): Promise<BatchPredictionResponse> {
    const response = await apiClient.post<BatchPredictionResponse>('/predict_batch', {
      transactions,
    });
    return response.data;
  },

  // Get model information and features
  async getModelInfo(): Promise<ModelInfo> {
    const response = await apiClient.get<ModelInfo>('/model_info');
    return response.data;
  },
};

export default fraudApi;
```

Create src/hooks/useFraudPrediction.ts:

```typescript
import { useState, useCallback } from 'react';
import { Transaction, PredictionResponse, ApiError } from '../types/fraud';
import fraudApi from '../services/fraudApi';

interface UseFraudPredictionReturn {
  predict: (transaction: Transaction) => Promise<void>;
  result: PredictionResponse | null;
  loading: boolean;
  error: ApiError | null;
  clearResult: () => void;
}

export const useFraudPrediction = (): UseFraudPredictionReturn => {
  const [result, setResult] = useState<PredictionResponse | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<ApiError | null>(null);

  const predict = useCallback(async (transaction: Transaction) => {
    setLoading(true);
    setError(null);
    setResult(null);

    try {
      const data = await fraudApi.predict(transaction);
      setResult(data);
    } catch (err) {
      setError(err as ApiError);
    } finally {
      setLoading(false);
    }
  }, []);

  const clearResult = useCallback(() => {
    setResult(null);
    setError(null);
  }, []);

  return { predict, result, loading, error, clearResult };
};

export default useFraudPrediction;
```

```typescript
import { useState, useCallback } from 'react';
import { Transaction, BatchPredictionResponse, ApiError } from '../types/fraud';
import fraudApi from '../services/fraudApi';

interface UseBatchPredictionReturn {
  predictBatch: (transactions: Transaction[]) => Promise<void>;
  result: BatchPredictionResponse | null;
  loading: boolean;
  error: ApiError | null;
  progress: number; // 0-100
}

export const useBatchPrediction = (): UseBatchPredictionReturn => {
  const [result, setResult] = useState<BatchPredictionResponse | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<ApiError | null>(null);
  const [progress, setProgress] = useState(0);

  const predictBatch = useCallback(async (transactions: Transaction[]) => {
    setLoading(true);
    setError(null);
    setProgress(0);

    try {
      // Simulate progress bar (updates every 100ms up to 90%)
      const progressInterval = setInterval(() => {
        setProgress((prev) => Math.min(prev + 10, 90));
      }, 100);

      const data = await fraudApi.predictBatch(transactions);
      
      clearInterval(progressInterval);
      setProgress(100);
      setResult(data);
    } catch (err) {
      setError(err as ApiError);
    } finally {
      setLoading(false);
      setProgress(0);
    }
  }, []);

  return { predictBatch, result, loading, error, progress };
};

export default useBatchPrediction;
```

Create src/components/BatchUpload.tsx:

```typescript
import React, { useState } from 'react';
import {
  Box,
  Button,
  Card,
  CardContent,
  LinearProgress,
  Table,
  TableBody,
  TableCell,
  TableContainer,
  TableHead,
  TableRow,
  Paper,
  Chip,
  Alert,
  Grid,
} from '@mui/material';
import { Transaction, BatchPredictionResponse } from '../types/fraud';
import useBatchPrediction from '../hooks/useBatchPrediction';

const BatchUpload: React.FC = () => {
  const { predictBatch, result, loading, error, progress } = useBatchPrediction();
  const [transactions, setTransactions] = useState<Transaction[]>([]);

  // Parse CSV file and extract transactions
  const handleFileUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;

    const reader = new FileReader();
    reader.onload = (event) => {
      try {
        const csv = event.target?.result as string;
        const lines = csv.split('\n').filter((line) => line.trim());
        const headers = lines[0].split(',').map((h) => h.trim());

        const data: Transaction[] = lines.slice(1).map((line) => {
          const values = line.split(',').map((v) => v.trim());
          return {
            TransactionAmt: parseFloat(values[0]),
            ProductCD: values[1] as Transaction['ProductCD'],
            DayOfWeek: parseInt(values[2]),
            Hour: parseInt(values[3]),
            CardType: values[4] as Transaction['CardType'],
            DeviceType: values[5] as Transaction['DeviceType'],
            OS: values[6] as Transaction['OS'],
            Browser: values[7] as Transaction['Browser'],
            Country: values[8],
            Merchant: values[9],
            Distance_km: parseFloat(values[10]),
            DaysSincePreviousTxn: parseFloat(values[11]),
            NumPreviousTxns: parseInt(values[12]),
          };
        });

        setTransactions(data);
      } catch (err) {
        console.error('CSV parsing error:', err);
        alert('Error parsing CSV file');
      }
    };
    reader.readAsText(file);
  };

  const handlePredict = async () => {
    if (transactions.length === 0) {
      alert('Please upload a CSV file with transactions first');
      return;
    }
    await predictBatch(transactions);
  };

  const downloadResults = () => {
    if (!result) return;

    const csv = [
      ['Index', 'Prediction', 'Fraud %', 'Confidence %', 'Risk Level'],
      ...result.predictions.map((pred, idx) => [
        idx + 1,
        pred.prediction === 1 ? 'FRAUD' : 'LEGITIMATE',
        (pred.fraud_probability * 100).toFixed(2),
        (pred.confidence * 100).toFixed(2),
        pred.risk_level,
      ]),
    ]
      .map((row) => row.join(','))
      .join('\n');

    const element = document.createElement('a');
    element.setAttribute('href', 'data:text/plain;charset=utf-8,' + encodeURIComponent(csv));
    element.setAttribute('download', 'fraud_predictions.csv');
    element.style.display = 'none';
    document.body.appendChild(element);
    element.click();
    document.body.removeChild(element);
  };

  return (
    <Card>
      <CardContent>
        <h2>📁 Batch Fraud Detection</h2>
        <p style={{ color: '#666' }}>Upload a CSV file to process multiple transactions at once</p>

        {/* File Upload Section */}
        <Box sx={{ mb: 3, p: 2, bgcolor: '#f5f5f5', borderRadius: 2 }}>
          <input
            type="file"
            accept=".csv"
            onChange={handleFileUpload}
            style={{ display: 'block', marginBottom: '10px' }}
          />
          <small style={{ color: '#999' }}>
            CSV format: TransactionAmt, ProductCD, DayOfWeek, Hour, CardType, DeviceType, OS, Browser,
            Country, Merchant, Distance_km, DaysSincePreviousTxn, NumPreviousTxns
          </small>
          <p style={{ marginTop: '10px' }}>
            <strong>{transactions.length}</strong> transactions loaded
          </p>
        </Box>

        {/* Predict Button */}
        <Button
          variant="contained"
          color="primary"
          onClick={handlePredict}
          disabled={loading || transactions.length === 0}
          fullWidth
          size="large"
        >
          {loading ? `Processing ${progress}%...` : 'Predict'}
        </Button>

        {/* Progress Bar */}
        {loading && (
          <Box sx={{ mt: 3 }}>
            <LinearProgress variant="determinate" value={progress} sx={{ height: 8, borderRadius: 4 }} />
            <p style={{ textAlign: 'center', marginTop: '10px', fontWeight: 'bold' }}>

        {/* Results Summary */}
        {result && (

                <TableBody>
                  {result.predictions.slice(0, 20).map((pred, idx) => (
```

```javascript
const API_BASE_URL = process.env.REACT_APP_API_URL || 'http://127.0.0.1:8000';
```

```javascript
import { useState, useCallback } from 'react';
import fraudApi from '../services/fraudApi';

export const useFraudPrediction = () => {
  const [result, setResult] = useState(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  const predict = useCallback(async (transaction) => {
    setLoading(true);
    setError(null);

    try {
      const data = await fraudApi.predict(transaction);
      setResult(data);
    } catch (err) {
      setError(err);
    } finally {
      setLoading(false);
    }
  }, []);

  return { predict, result, loading, error };
};
```
